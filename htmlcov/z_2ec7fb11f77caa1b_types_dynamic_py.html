154 def from_simple_type(self, value): 

155 """ 

156 Convert a simple type representation into an instance of the model 

157 type. 

158 

159 This function takes a dictionary representation of the model and 

160 converts it into an instance of the associated model type. Each field 

161 in the input dictionary is processed using the corresponding 

162 property's `from_simple_type` method. If there are any extra fields 

163 in the input dictionary that do not correspond to the model's 

164 properties, a `ValueError` is raised. 

165 

166 :param value: A dictionary where keys are field names and values are 

167 the simple type representations of the model's properties. 

168 :return: An instance of the model type with fields populated from the 

169 input dictionary. 

170 :raises ValueError: If the input dictionary contains unknown fields. 

171 """ 

172 

173 copied_value = copy.deepcopy(value) 

174 parsed_value = {} 

175 for name, prop in self._python_type.properties.properties.items(): 

176 if name in copied_value: 

177 property_type = prop.get_property_type() 

178 try: 

179 val = copied_value.pop(name) 

180 parsed_value[name] = property_type.from_simple_type(val) 

181 except (ValueError, TypeError): 

182 raise ra_exc.ParseError(value=f"{name}={val}") 

183 except ra_exc.ParseError as e: 

184 raise ra_exc.ParseError(value=f"{name}={e.value}") 

185 if copied_value: 

186 raise ra_exc.ParseError(value=f"(Unknown fields: {copied_value})") 

187 return self._python_type(**parsed_value)

284 def from_simple_type(self, value): 

285 """ 

286 Convert a simple type representation into an instance of the model 

287 type. 

288 

289 This function takes a dictionary representation of the model and 

290 determines the appropriate model class to use based on the 'kind' 

291 field. It then uses the `from_simple_type` method of the identified 

292 model class to instantiate an instance of the model type. If the 

293 'kind' is not found or is invalid, an `UnknownType` error is raised. 

294 

295 :param value: A dictionary with a 'kind' field indicating the model 

296 type and additional fields representing the simple type data. 

297 :return: An instance of the model type corresponding to the 'kind' 

298 field. 

299 :raises UnknownType: If the 'kind' field does not correspond to any 

300 known model type. 

301 """ 

302 try: 

303 value_type_name = value["kind"] 

304 value_class = self._kind_type_map[value_type_name] 

305 return value_class.from_simple_type(value) 

306 except (AttributeError, KeyError): 

307 raise UnknownType(value=value)
